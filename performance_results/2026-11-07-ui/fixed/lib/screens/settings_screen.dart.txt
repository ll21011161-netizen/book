class SettingsScreen extends StatefulWidget {
  const SettingsScreen({super.key});

  @override
  State<SettingsScreen> createState() => _SettingsScreenState();
}

    with AutomaticKeepAliveClientMixin {
  @override
  bool get wantKeepAlive => true;

  @override
  Widget build(BuildContext context) {
    super.build(context);
    final settings = context.watch<SettingsProvider>();
    final theme = Theme.of(context);
    final colors = theme.colorScheme;

    return Scaffold(
      body: SafeArea(
        child: Center(
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 700),
            child: ListView(
              padding: const EdgeInsets.fromLTRB(20, 24, 20, 116),
              children: [
                Row(
                  children: [
                    Expanded(
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          Text('設定',
                              style: theme.textTheme.headlineMedium?.copyWith(
                                fontWeight: FontWeight.w800,
                              )),
                          const SizedBox(height: 5),
                          Text(
                            '把書架與閱讀方式調整成你喜歡的樣子',
                            style: theme.textTheme.bodyMedium?.copyWith(
                              color: colors.onSurfaceVariant,
                            ),
                          ),
                        ],
                      ),
                    ),
                    CircleAvatar(
                      radius: 26,
                      backgroundColor: colors.primaryContainer,
                      child: Icon(Icons.tune_rounded,
                          color: colors.onPrimaryContainer),
                    ),
                  ],
                ),
                const SizedBox(height: 28),
                _SettingsSection(
                  icon: Icons.grid_view_rounded,
                  title: '書架與封面',
                  description: '調整漫畫在書架上的排列方式',
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Text('每列顯示', style: theme.textTheme.titleMedium),
                      const SizedBox(height: 4),
                      Text(
                        '選擇封面大小，3 欄最大、5 欄最精簡',
                        style: theme.textTheme.bodySmall?.copyWith(
                          color: colors.onSurfaceVariant,
                        ),
                      ),
                      const SizedBox(height: 14),
                      Wrap(
                        spacing: 8,
                        runSpacing: 8,
                        children: [
                          for (final columns in const [3, 4, 5])
                            ChoiceChip(
                              label: Text('$columns 欄'),
                              selected: settings.bookshelfColumns == columns,
                              onSelected: (_) =>

                        ],
                      ),
                      const Divider(height: 32),
                      Text('漫畫封面顯示', style: theme.textTheme.titleMedium),
                      const SizedBox(height: 4),
                      Text(
                        '預設填滿：所有封面大小一致，可能裁切邊緣',

                          color: colors.onSurfaceVariant,

                      const SizedBox(height: 14),
                      Wrap(
                        spacing: 8,
                        runSpacing: 8,
                        children: [
                          ChoiceChip(
                            avatar: const Icon(Icons.crop_rounded, size: 18),
                            label: const Text('填滿封面'),
                            selected: settings.cropBookCovers,
                            onSelected: (_) => settings.setCropBookCovers(true),

                            avatar:

                    title: const Text('深色模式'),
                    subtitle: const Text('適合在光線較暗的環境使用'),
                    value: settings.darkMode,
                    onChanged: settings.setDarkMode,

                ),
                const SizedBox(height: 22),

                  icon: Icons.menu_book_rounded,

                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      SwitchListTile(

class _SettingsSection extends StatelessWidget {
  const _SettingsSection({
    required this.icon,
    required this.title,
    required this.description,
    required this.child,
  });

  final IconData icon;
  final String title;
  final String description;
  final Widget child;

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final colors = theme.colorScheme;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Row(
          children: [
            Icon(icon, color: colors.primary, size: 22),
            const SizedBox(width: 10),
            Text(title,
                style: theme.textTheme.titleLarge?.copyWith(
                  fontWeight: FontWeight.w800,
                )),
          ],
        ),
        const SizedBox(height: 3),
        Padding(
          padding: const EdgeInsets.only(left: 32),
          child: Text(description,
              style: theme.textTheme.bodySmall?.copyWith(
                color: colors.onSurfaceVariant,
              )),
        ),
        const SizedBox(height: 12),
        Container(
          width: double.infinity,
          padding: const EdgeInsets.symmetric(horizontal: 18, vertical: 14),
          decoration: BoxDecoration(
            color: colors.surfaceContainerLow,
            borderRadius: BorderRadius.circular(24),
          ),
          child: child,
        ),
      ],
    );
  }
}

class _ZoomSetting extends StatefulWidget {
  const _ZoomSetting();
  @override
  State<_ZoomSetting> createState() => _ZoomSettingState();
}

class _ZoomSettingState extends State<_ZoomSetting> {
  double? _zoomDraft;
  @override
  Widget build(BuildContext context) {
    final savedZoom =
        context.select<SettingsProvider, double>((s) => s.zoomLevel);
    final settings = context.read<SettingsProvider>();
    final theme = Theme.of(context);
    final colors = theme.colorScheme;
    final zoom = _zoomDraft ?? savedZoom;
    return Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
      Row(
        children: [
          const Icon(Icons.zoom_in_rounded),
          const SizedBox(width: 12),
          Expanded(child: Text('雙擊放大倍率', style: theme.textTheme.titleMedium)),
          Text(
            '${zoom.toStringAsFixed(1)}×',
            style: theme.textTheme.titleMedium?.copyWith(
              color: colors.primary,
              fontWeight: FontWeight.w800,
            ),
          ),
        ],
      ),
      const SizedBox(height: 4),
      Text(
        '雙擊圖片時使用此倍率；雙指仍可自由縮放。',
        style: theme.textTheme.bodySmall?.copyWith(
          color: colors.onSurfaceVariant,
        ),
      ),
      Slider(
        value: zoom.clamp(AppConfig.zoomMin, AppConfig.zoomMax).toDouble(),
        min: AppConfig.zoomMin,
        max: AppConfig.zoomMax,
        divisions: AppConfig.zoomDivisions,
        label: '${zoom.toStringAsFixed(1)}×',
        onChanged: (value) => setState(() => _zoomDraft = value),
        onChangeEnd: (value) {
          settings.setZoomLevel(value);
          setState(() => _zoomDraft = null);
        },
      ),
      Row(
        mainAxisAlignment: MainAxisAlignment.spaceBetween,
        children: [
          Text('${AppConfig.zoomMin.toStringAsFixed(1)}×',
              style: theme.textTheme.bodySmall),
          TextButton.icon(
            onPressed: () {
              settings.setZoomLevel(2.5);
              setState(() => _zoomDraft = null);
            },
            icon: const Icon(Icons.restart_alt_rounded, size: 18),
            label: const Text('恢復預設 2.5×'),
          ),
          Text('${AppConfig.zoomMax.toStringAsFixed(1)}×',
              style: theme.textTheme.bodySmall),
        ],
      ),
    ]);
  }
}
