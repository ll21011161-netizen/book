import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:gun_book/core/constants.dart';
import 'package:gun_book/models/library_item.dart';
import 'package:gun_book/widgets/book_card.dart';
import 'package:gun_book/widgets/folder_card.dart';
import 'package:gun_book/widgets/page_thumbnail.dart';
import 'package:gun_book/widgets/spring_press.dart';
import 'package:gun_book/widgets/smooth_input_dialog.dart';

void main() {
  for (final folder in [false, true]) {
    for (final reduced in [false, true]) {
      testWidgets(
          '20 presses, cancel and long press folder=$folder reduced=$reduced',
          (tester) async {
        var taps = 0;
        var holds = 0;
        final card = folder
            ? FolderCard(
                folder: MangaFolder(id: 'folder', title: 'folder'),
                onTap: () => taps++,
                onLongPress: () => holds++)
            : BookCard(
                book: MangaBook(id: 'book', title: 'book', filePaths: []),
                onTap: () => taps++,
                onLongPress: () => holds++);
        await tester.pumpWidget(MaterialApp(
            home: MediaQuery(
                data: MediaQueryData(disableAnimations: reduced),
                child: Center(
                    child: SizedBox(width: 110, height: 210, child: card)))));
        await tester.pumpAndSettle();
        final press = find.byType(SpringPress);
        final thumbnails =
            tester.elementList(find.byType(PageThumbnail)).toList();
        final size = tester.getSize(press);
        for (var i = 0; i < 20; i++) {
          final gesture = await tester.startGesture(tester.getCenter(press));
          await tester.pump();
          // Pointer feedback starts before GestureDetector's 100ms tap timeout.
          expect(
              tester
                  .widget<AnimatedScale>(find.byType(AnimatedScale).first)
                  .scale,
              reduced ? 1 : AppConstants.pressScale);
          await tester.pump(AppConstants.pressDuration);
          final scale =
              tester.widget<AnimatedScale>(find.byType(AnimatedScale).first);
          expect(scale.scale, reduced ? 1 : AppConstants.pressScale);
          await gesture.up();
          await tester.pump(const Duration(milliseconds: 15));
        }
        await tester.pumpAndSettle();
        expect(taps, 20);
        expect(tester.getSize(press), size);
        expect(tester.elementList(find.byType(PageThumbnail)).toList(),
            thumbnails);
        final cancel = await tester.startGesture(tester.getCenter(press));
        await tester.pump(const Duration(milliseconds: 120));
        await cancel.cancel();
        await tester.pumpAndSettle();
        expect(
            tester
                .widget<AnimatedScale>(find.byType(AnimatedScale).first)
                .scale,
            1);
        await tester.longPress(press);
        await tester.pumpAndSettle();
        expect(holds, 1);
        expect(
            tester
                .widget<AnimatedScale>(find.byType(AnimatedScale).first)
                .scale,
            1);
        expect(tester.takeException(), isNull);
      });
    }
  }

  for (final reduced in [false, true]) {
    testWidgets(
        'dialog focus waits for entrance; 10 dismissal cycles reduced=$reduced',
        (tester) async {
      final nav = GlobalKey<NavigatorState>();
      String? result;
      await tester.pumpWidget(MaterialApp(
          navigatorKey: nav,
          builder: (context, child) => MediaQuery(
              data: MediaQuery.of(context).copyWith(disableAnimations: reduced),
              child: child!),
          home: Builder(
              builder: (context) => Scaffold(
                  body: TextButton(
                      onPressed: () async {
                        result = await SmoothInputDialog.show(context,
                            title: 'Name', initialText: 'original');
                      },
                      child: const Text('Open'))))));
      for (var i = 0; i < 10; i++) {
        await tester.tap(find.text('Open'));
        await tester.pump();
        final field = tester.widget<TextField>(find.byType(TextField));
        if (!reduced) expect(field.focusNode!.hasFocus, isFalse);
        if (i == 0) {
          // Dismiss during entrance: no delayed keyboard/focus callback remains.
          nav.currentState!.pop();
        } else {
          await tester.pumpAndSettle();
          expect(field.focusNode!.hasFocus, isTrue);
          await tester.enterText(find.byType(TextField), 'updated');
          switch (i % 4) {
            case 0:
              await tester.tap(find.byType(ElevatedButton));
              break;
            case 1:
              await tester.tap(find.byType(TextButton).last);
              break;
            case 2:
              await tester.tapAt(const Offset(5, 5));
              break;
            case 3:
              await tester.binding.handlePopRoute();
              break;
          }
        }
        await tester.pumpAndSettle();
        expect(find.byType(SmoothInputDialog), findsNothing);
        expect(result, i > 0 && i % 4 == 0 ? 'updated' : null);
        expect(tester.takeException(), isNull);
      }
      await tester.tap(find.text('Open'));
      await tester.pump(const Duration(milliseconds: 30));
      await tester.pumpWidget(const SizedBox());
      await tester.pumpAndSettle();
      expect(tester.takeException(), isNull);
    });
  }
}
