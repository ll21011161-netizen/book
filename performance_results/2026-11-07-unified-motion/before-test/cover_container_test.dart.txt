Rect frame(WidgetTester tester) {
  final position = tester.widget<Positioned>(find
      .descendant(
          of: find.byType(CoverContainerTransition),
          matching: find.byType(Positioned))
      .first);
  return Rect.fromLTWH(
      position.left!, position.top!, position.width!, position.height!);
}

Finder flightOpacity() => find
    .descendant(
        of: find.byType(CoverContainerTransition),
        matching: find.byType(Opacity))
    .first;

void main() {
  test('detail preference persists, unknown and old values keep page behavior',
      () async {
    expect(MotionSettings.fromJson({}).detail, DetailMotion.followPage);
    expect(MotionSettings.fromJson({'detail': 'future'}).detail,
        DetailMotion.followPage);
    SharedPreferences.setMockInitialValues({});
    final settings = SettingsProvider();
    settings.setMotion(
        const MotionSettings(detail: DetailMotion.containerTransform));
    await settings.flushMotionSettings();
    final next = SettingsProvider();
    while (!next.isLoaded) {
      await Future<void>.value();
    }
    expect(next.motion.detail, DetailMotion.containerTransform);
    settings.dispose();
    next.dispose();
  });

      for (final fit in [BoxFit.cover, BoxFit.contain]) {
        testWidgets('real anchor $folder $position $fit expands and returns',
            (tester) async {
          final nav = GlobalKey<NavigatorState>();
          late BuildContext home;
          CoverOrigin? source;
          DetailRoute<void>? route;
          var builds = 0;
          final card = folder
              ? FolderCard(
                  folder: MangaFolder(id: 'folder', title: 'Folder'),
                  onTap: () {},
                  onLongPress: () {},
                  onOpen: (s) {

                  onTap: () {},
                  onLongPress: () {},
                  onOpen: (s) {
                    source = s;
                  });
          await tester.pumpWidget(MaterialApp(
              navigatorKey: nav,
              builder: (_, child) => MotionScope(
                  settings: const MotionSettings(
                      detail: DetailMotion.containerTransform),
                  child: child!),
              home: Builder(builder: (context) {
                home = context;
                builds++;
                return Scaffold(
                    body: Stack(children: [

                      left: position.dx,
                      top: position.dy,
                      width: 130,
                      height: 230,
                      child: card)
                ]));
              })));
          await tester.pumpAndSettle();
          await tester.tap(find.byType(SpringPress));
          await tester.pumpAndSettle();
          final start = source!.resolveRect()!;
          expect(start.topLeft, position);
          // Cover-only boundary excludes title/progress labels.
          expect(start.height, lessThan(230));
          final snapshot = source!.captureImage();
          expect(snapshot, isNotNull);
          expect(snapshot!.width, lessThanOrEqualTo(512));
          snapshot.dispose();
          route = DetailRoute<void>(
              context: home,
              source: source,
              page:
                  const Scaffold(body: Center(child: Text('Detail content'))));
          expect(route.hasCoverTransform, isTrue);
          expect(route.transitionDuration, const Duration(milliseconds: 380));
          expect(route.reverseTransitionDuration,
              const Duration(milliseconds: 300));
          nav.currentState!.push(route);
          await tester.pump();
          await tester.pump();
          expect(frame(tester), start);
          final detailElement =
              tester.element(find.text('Detail content', skipOffstage: false));
          await tester.pump(const Duration(milliseconds: 140));
          final middle = frame(tester);
          expect(middle.width, greaterThan(start.width));
          expect(middle.height, greaterThan(start.height));
          await tester.pumpAndSettle();

          expect(source!.resolveRect(), start);
          expect(tester.takeException(), isNull);

  testWidgets(
      'moved target resolves again; predictive back cancels and commits',
      (tester) async {
    final nav = GlobalKey<NavigatorState>();
    late BuildContext home;
    Rect? target = const Rect.fromLTWH(25, 40, 120, 180);
    final origin = CoverOrigin(
        resolveRect: () => target,
        captureImage: () => null,
        radius: 20,
        color: Colors.blue);
    await tester.pumpWidget(MaterialApp(
        navigatorKey: nav,
        theme: ThemeData(platform: TargetPlatform.android),
        builder: (_, child) => MotionScope(
            settings:
                const MotionSettings(detail: DetailMotion.containerTransform),
            child: child!),
        home: Builder(builder: (c) {
          home = c;
          return const Text('home');
        })));
    final route = DetailRoute<void>(
        context: home, source: origin, page: const Text('detail'));
    nav.currentState!.push(route);
    await tester.pumpAndSettle();
    Future<dynamic> event(String name, [double? progress]) async {
      final result = Completer<dynamic>();
      await tester.binding.defaultBinaryMessenger.handlePlatformMessage(
          SystemChannels.backGesture.name,
          SystemChannels.backGesture.codec.encodeMethodCall(MethodCall(
              name,
              progress == null
                  ? null
                  : {
                      'progress': progress,
                      'touchOffset': [8.0, 250.0],
                      'swipeEdge': 0
                    })),
          (reply) => result.complete(
              SystemChannels.backGesture.codec.decodeEnvelope(reply!)));
      return result.future;
    }

    target = const Rect.fromLTWH(400, 220, 120, 180);
    expect(await event('startBackGesture', 0), isTrue);
    await tester.pump();
    await event('updateBackGestureProgress', .5);
    await tester.pump();
    expect(frame(tester).left, closeTo(200, .01));
    await event('cancelBackGesture');
    await tester.pumpAndSettle();
    expect(frame(tester), const Rect.fromLTWH(0, 0, 800, 600));
    expect(await event('startBackGesture', 0), isTrue);
    await tester.pump();
    await event('updateBackGestureProgress', .75);
    await tester.pump();
    final before = frame(tester);
    await event('commitBackGesture');
    await tester.pump();
    expect((frame(tester).topLeft - before.topLeft).distance, lessThan(.1));
    await tester.pumpAndSettle();
    expect(find.text('detail'), findsNothing);
    expect(tester.takeException(), isNull);
  });

  testWidgets('capture painted container frames without manga decoding',
      (tester) async {
    final nav = GlobalKey<NavigatorState>();
    final scene = GlobalKey();
    final cover = GlobalKey<CoverHeroState>();
    late BuildContext home;
    await tester.pumpWidget(MaterialApp(
        navigatorKey: nav,
        builder: (_, child) => MotionScope(
            settings:
                const MotionSettings(detail: DetailMotion.containerTransform),
            child: RepaintBoundary(key: scene, child: child!)),
        home: Builder(builder: (c) {
          home = c;
          return Scaffold(
              body: Stack(children: [
            Positioned(
                left: 48,
                top: 160,
                width: 130,
                height: 190,
                child: CoverHero(
                    key: cover,
                    tag: 'painted_cover',
                    identity: 'painted',
                    builder: (_) => const ColoredBox(
                        color: Colors.indigo,
                        child: Center(
                            child: Icon(Icons.menu_book,
                                color: Colors.amber, size: 60)))))
          ]));
        })));
    await tester.pumpAndSettle();
    nav.currentState!.push(DetailRoute<void>(
        context: home,
        source: cover.currentState!.origin,
        page: Scaffold(
            appBar: AppBar(title: const Text('Detail')),
            body: GridView.count(
                crossAxisCount: 4,
                padding: const EdgeInsets.all(20),
                children: [
                  for (var i = 0; i < 12; i++)
                    const Card(child: Center(child: Icon(Icons.photo_outlined)))
                ]))));
    await tester.pump();
    await tester.pump();
    Future<void> capture(String name) async {
      await tester.runAsync(() async {
        final boundary =
            scene.currentContext!.findRenderObject() as RenderRepaintBoundary;
        final image = await boundary.toImage(pixelRatio: 1);
        final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
        final directory = Directory('build/container_preview')
          ..createSync(recursive: true);
        File('${directory.path}/$name.png')
            .writeAsBytesSync(bytes!.buffer.asUint8List());
        image.dispose();
      });
    }

    await capture('start');
    await tester.pump(const Duration(milliseconds: 90));
    await capture('expanding');
    await tester.pumpAndSettle();
    await capture('complete');
    nav.currentState!.pop();
    await tester.pump();
    await tester.pump(const Duration(milliseconds: 160));
    await capture('closing');
    await tester.pumpAndSettle();
    expect(tester.takeException(), isNull);
  });
